\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb,tikz}
\usetikzlibrary{arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0,>=latex,line cap=round]
  % centre at (C,0) with C = 1, X = 1.5
  \def\C{1}\def\X{1.5}
  \draw[->] (-0.3,0) -- (5.6,0) node[right]{$x$};
  \draw[->] (0,-2.9) -- (0,3.3) node[above]{$ct$};
  % asymptotes x - C = +-ct
  \draw[dashed,black!60] (\C-0.2,-0.2) -- (\C+3.1,3.1) node[above right,black,font=\small]{$x - C = ct$};
  \draw[dashed,black!60] (\C-0.2,0.2) -- (\C+2.9,-2.9) node[below right,black,font=\small]{$x - C = -ct$};
  % hyperbola worldline
  \draw[very thick,domain=-2.75:2.75,samples=80,variable=\t] plot ({\C+sqrt(\X*\X+\t*\t)},{\t});
  \fill (\C,0) circle (1.5pt) node[below left,font=\small]{$(C,0)$};
  \draw[<->,black!70] (\C,0.15) -- (\C+\X,0.15) node[midway,above,black,font=\small]{$X=c^2/a_0$};
  \fill (\C+\X,0) circle (1.5pt);
\end{tikzpicture}
\end{document}
